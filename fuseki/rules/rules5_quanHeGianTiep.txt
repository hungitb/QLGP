@prefix person: http://qlgp/person#
@prefix inferred: http://qlgp/inferred#
@prefix quanHeTrucTiep: http://qlgp/quanHeTrucTiep#
@prefix rdf: http://www.w3.org/1999/02/22-rdf-syntax-ns#
@prefix ttgp: http://qlgp/thongTinGiaPha#

# Số đời
[
  (:thongTinGiaPha ttgp:ancestor ?x)
  ->
  (?x inferred:doiThu "1"^^xsd:integer)
]

[
    (?x inferred:doiThu ?soDoiX),
    (?y person:hasFather ?x),
    addOne(?soDoiX, ?soDoiY)
    ->
    (?y inferred:doiThu ?soDoiY)
]

[
    (?x inferred:doiThu ?soDoiX),
    (?y person:hasMother ?x),
    addOne(?soDoiX, ?soDoiY)
    ->
    (?y inferred:doiThu ?soDoiY)
]

# Đời trên, chỉ xác định với những người thuộc gia phả (vì thuật toán xđ mqh dùng cái này cùng với số đời)
[
    (?x person:hasFather ?y),
    (:thongTinGiaPha ttgp:ancestor ?y)
    ->
    (?x inferred:doiTren ?y)
]

[
    (?x person:hasMother ?y),
    (:thongTinGiaPha ttgp:ancestor ?y)
    ->
    (?x inferred:doiTren ?y)
]

# Người có bố tồn tại đời trên thì mới có bố là đời trên
# Dùng để giới hạn chỉ trong gia phả
[
    (?x person:hasFather ?y),
    (?y inferred:doiTren ?z)
    ->
    (?x inferred:doiTren ?y)
]

# Tương tự như phần của bố
[
    (?x person:hasMother ?y),
    (?y inferred:doiTren ?z)
    ->
    (?x inferred:doiTren ?y)
]

[
    (?x inferred:doiTren ?y),
    (?y inferred:doiTren ?z)
    ->
    (?x inferred:doiTren ?z)
]
