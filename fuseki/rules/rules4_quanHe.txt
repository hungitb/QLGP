### Quan hệ

[
  (?x person:hasFather ?y)
  ->
  (?x quanHeTrucTiep:Bo ?y),
  [ ( ?y quanHeTrucTiep:ConTrai ?x ) <- (?x rdf:type :male) ],
  [ ( ?y quanHeTrucTiep:ConGai ?x ) <- (?x rdf:type :female) ]
]

[
  (?x person:hasMother ?y)
  ->
  (?x quanHeTrucTiep:Me ?y),
  [ ( ?y quanHeTrucTiep:ConTrai ?x ) <- (?x rdf:type :male) ],
  [ ( ?y quanHeTrucTiep:ConGai ?x ) <- (?x rdf:type :female) ]
]

[
  (?x person:hasFather ?z),
  (?y person:hasFather ?z),
  (?x person:lonTuoiHon ?y)
  ->
  [ ( ?y quanHeTrucTiep:AnhTrai ?x ) <- (?x rdf:type :male) ],
  [ ( ?y quanHeTrucTiep:ChiGai ?x ) <- (?x rdf:type :female) ],
  [ ( ?x quanHeTrucTiep:EmTrai ?y ) <- (?y rdf:type :male) ],
  [ ( ?x quanHeTrucTiep:EmGai ?y ) <- (?y rdf:type :female) ]
]

[
  (?x person:hasMother ?z),
  (?y person:hasMother ?z),
  (?x person:lonTuoiHon ?y)
  ->
  [ ( ?y quanHeTrucTiep:AnhTrai ?x ) <- (?x rdf:type :male) ],
  [ ( ?y quanHeTrucTiep:ChiGai ?x ) <- (?x rdf:type :female) ],
  [ ( ?x quanHeTrucTiep:EmTrai ?y ) <- (?y rdf:type :male) ],
  [ ( ?x quanHeTrucTiep:EmGai ?y ) <- (?y rdf:type :female) ]
]
