@prefix : http://qlgp#
@prefix person: http://qlgp/person#
@prefix rdf: http://www.w3.org/1999/02/22-rdf-syntax-ns#
@prefix ttgp: http://qlgp/thongTinGiaPha#

### Thuộc gia phả

# Tổ tiên thuộc gia phả
[
  (:thongTinGiaPha ttgp:ancestor ?x)
  ->
  (?x person:thuocGiaPha "true")
]

# Bạn đời của một Nam trong gia phả thuộc chế độ phả hệ sẽ thuộc gia phả do quan hệ vợ chồng
[
  (?x person:thuocGiaPha "true"),
  (?x rdf:type :male),
  (:thongTinGiaPha ttgp:type "phaHe"),
  (?x person:hasSpouse ?y)
  ->
  (?y person:thuocGiaPhaDoVoChong "true")
]

# Bạn đời của một Nữ trong gia phả thuộc chế độ mẫu hệ sẽ thuộc gia phả do quan hệ vợ chồng
[
  (?x person:thuocGiaPha "true"),
  (?x rdf:type :female),
  (:thongTinGiaPha ttgp:type "mauHe"),
  (?x person:hasSpouse ?y)
  ->
  (?y person:thuocGiaPhaDoVoChong "true")
]

# Trường hợp đặc biệt của tổ tiên, dù là nam hay nữ, dù là phả hệ hay mẫu hệ thì bạn đời cũng sẽ thuộc gia phả
[
  (:thongTinGiaPha ttgp:ancestor ?x),
  (?x person:hasSpouse ?y)
  ->
  (?y person:thuocGiaPhaDoVoChong "true")
]

# Người thuộc gia phả do quan hệ vợ chồng thì cũng thuộc gia phả
[
  (?x person:thuocGiaPhaDoVoChong "true")
  ->
  (?x person:thuocGiaPha "true")
]

# Trong gia tộc theo chế độ phả hệ, người có bố thuộc gia phả cũng thuộc gia phả do quan hệ ruột thịt
# noValue: Bỏ qua con riêng
[
  (?x person:thuocGiaPha "true"),
  (:thongTinGiaPha ttgp:type "phaHe"),
  (?y person:hasFather ?x),
  noValue(?x person:thuocGiaPhaDoVoChong "true")
  ->
  (?y person:thuocGiaPha "true")
]

# Trong gia tộc theo chế độ mẫu hệ, người có mẹ thuộc gia phả cũng thuộc gia phả do quan hệ ruột thịt
# noValue: Bỏ qua con riêng
[
  (?x person:thuocGiaPha "true"),
  (:thongTinGiaPha ttgp:type "mauHe"),
  (?y person:hasMother ?x),
  noValue(?x person:thuocGiaPhaDoVoChong "true")
  ->
  (?y person:thuocGiaPha "true")
]
